# Plan: KAN-251 / KAN-252 / KAN-253 — Data Layer Refactor

**Phase:** Refactor (Tier 2 items #4–#7 from `docs/plans/refactor_plan.md`)
**Jira:** KAN-251, KAN-252, KAN-253
**Depends on:** none (all three touch `data/` and `models/` — no feature-level dependencies)

## Overview

This combined plan implements three related refactoring tickets from the architectural review's Tier 2 items. Together they introduce proper abstraction seams in the data layer:

1. **KAN-253** — Move `GRAIN_TO_TIME_UNIT` to `shelves/schema/temporal.py`, wrap loader IO errors with file-path context, and introduce `NoDataSourceError` in `data/bind.py`.
2. **KAN-252** — Extract an `HTTPTransport` protocol from `cube_client.py`, classify HTTP errors into `CubeAuthError` / `CubeServerError` / `CubeTimeoutError`, and truncate error bodies.
3. **KAN-251** — Introduce a `DataSourceAdapter` protocol and registry in `data/`, move Cube fetch logic behind it, and have `bind.py` do a registry lookup instead of an `if` branch.

Implementation order is 253 → 252 → 251 because each step builds on the prior: KAN-253 cleans up shared constants and error types, KAN-252 adds the transport seam that the adapter will use, and KAN-251 wires the adapter protocol on top.

**Note:** The refactor plan (§ 2.5) flags `GRAIN_TO_TIME_UNIT` as duplicated in both `resolver.py` and `cube_client.py`. As of the current codebase, the duplication has already been resolved — the constant only exists in `resolver.py:26-32`. The move to `schema/temporal.py` is still valuable for cohesion (it's a schema-level mapping, not a model-resolver concern), but there is no second copy to delete from `cube_client.py`.

### `shelves/data/cube_client.py` — new/modified types and functions

```python
# ─── New error subclasses ────────────────────────────────────────

class CubeAuthError(CubeError):
    """Cube API returned 401 or 403."""

class CubeServerError(CubeError):
    """Cube API returned 5xx."""

class CubeTimeoutError(CubeError):
    """Cube request timed out."""


# ─── HTTPTransport protocol ─────────────────────────────────────

class HTTPTransport(Protocol):
    def post(self, url: str, *, json: dict, headers: dict) -> httpx.Response: ...


class HttpxTransport:
    """Default transport using httpx.Client."""

    def __init__(self, timeout: float = 30.0) -> None:
        self._timeout = timeout

    def post(self, url: str, *, json: dict, headers: dict) -> httpx.Response:
        with httpx.Client(timeout=self._timeout) as client:
            return client.post(url, json=json, headers=headers)


# ─── Modified fetch_from_cube_model ──────────────────────────────

def fetch_from_cube_model(
    model: DataModel,
    chart_spec: ChartSpec,
    resolver: FieldTypeResolver,
    config: CubeConfig | None = None,
    transport: HTTPTransport | None = None,
) -> list[dict[str, Any]]:
    """
    Fetch data from a Cube.dev instance using a DataModel.

    Args:
        model: The loaded DataModel (must have a CubeSource).
        chart_spec: The parsed ChartSpec (for field extraction + filters).
        resolver: ModelResolver for classifying fields.
        config: Cube connection config. If None, reads from environment.
        transport: HTTP transport. If None, uses HttpxTransport (default httpx).

    Returns:
        List of row dicts with field names matching the DSL (no cube prefix).

    Raises:
        CubeConfigError: If environment variables are not set.
        CubeAuthError: If Cube returns 401 or 403.
        CubeServerError: If Cube returns 5xx.
        CubeTimeoutError: If the request times out.
        CubeQueryError: If Cube returns any other non-200 status.
    """
    # Step 1: Resolve config
    if config is None:
        config = CubeConfig.from_env()

    # Step 2: Resolve transport
    if transport is None:
        transport = HttpxTransport()

    # Step 3: Build query (unchanged)
    assert isinstance(model.source, CubeSource)
    cube_name = model.source.cube
    query = build_cube_query(cube_name, chart_spec, resolver)

    # Step 4: Execute HTTP request with error classification
    url = f"{config.api_url}/cubejs-api/v1/load"
    headers = {"Authorization": config.api_token}

    try:
        response = transport.post(url, json={"query": query}, headers=headers)
    except httpx.TimeoutException as e:
        raise CubeTimeoutError(f"Cube request timed out: {e}") from e

    # Step 5: Classify HTTP errors
    if response.status_code != 200:
        body = response.text[:500]  # truncate large error pages
        if response.status_code in (401, 403):
            raise CubeAuthError(
                f"Cube auth error (HTTP {response.status_code}): {body}"
            )
        elif response.status_code >= 500:
            raise CubeServerError(
                f"Cube server error (HTTP {response.status_code}): {body}"
            )
        else:
            raise CubeQueryError(
                f"Cube API error (HTTP {response.status_code}): {body}"
            )

    # Step 6: Parse and strip prefixes (unchanged)
    raw_rows = response.json().get("data", [])
    return [_strip_prefix(row) for row in raw_rows]
```

```python
from __future__ import annotations

from typing import Any, Protocol

from shelves.schema.chart_schema import ChartSpec
from shelves.models.schema import DataModel
from shelves.schema.field_types import FieldTypeResolver


class DataSourceAdapter(Protocol):
    """Protocol for fetching data from a source configured in a DataModel."""

    def fetch(
        self,
        model: DataModel,
        chart_spec: ChartSpec,
        resolver: FieldTypeResolver,
    ) -> list[dict[str, Any]]:
        """Fetch rows from the data source."""
        ...


_registry: dict[str, DataSourceAdapter] = {}


def register(source_type: str, adapter: DataSourceAdapter) -> None:
    """Register an adapter for a data source type (e.g. "cube")."""
    _registry[source_type] = adapter


def get_adapter(source_type: str) -> DataSourceAdapter:
    """
    Look up a registered adapter by source type.

    Raises KeyError if no adapter is registered for the given type.
    """
    if source_type not in _registry:
        raise KeyError(
            f"No data source adapter registered for type '{source_type}'. "
            f"Available: {sorted(_registry.keys())}"
        )
    return _registry[source_type]


class CubeDataSourceAdapter:
    """Adapter that fetches data from Cube.dev via cube_client."""

    def fetch(
        self,
        model: DataModel,
        chart_spec: ChartSpec,
        resolver: FieldTypeResolver,
    ) -> list[dict[str, Any]]:
        from shelves.data.cube_client import fetch_from_cube_model
        return fetch_from_cube_model(model, chart_spec, resolver)


# Self-register on import
register("cube", CubeDataSourceAdapter())
```

### `shelves/data/bind.py` — modified `resolve_data`

```python
def resolve_data(
    spec: dict,
    chart_spec: ChartSpec,
    rows: list[dict] | None = None,
    models_dir: str | Path | None = None,
) -> dict:
    """
    Attach data to a Vega-Lite spec, using inline rows or a registered adapter.

    ...same docstring, updated Raises section...

    Raises:
        NoDataSourceError: If no rows provided and the model has no supported source.
    """
    if rows is not None:
        return bind_data(spec, rows)

    from shelves.models.loader import load_model
    from shelves.models.resolver import ModelResolver

    model = load_model(chart_spec.data, models_dir=models_dir)
    resolver = ModelResolver(model)

    if model.source:
        from shelves.data.sources import get_adapter
        try:
            adapter = get_adapter(model.source.type)
        except KeyError:
            pass
        else:
            fetched = adapter.fetch(model, chart_spec, resolver)
            return bind_data(spec, fetched)

    from shelves.data.errors import NoDataSourceError
    raise NoDataSourceError(
        f"No data provided for model '{chart_spec.data}'. "
        "Pass --data or configure a supported source in the model."
    )
```

### `shelves/models/loader.py` — modified `load_model`

```python
def load_model(
    model_name: str,
    models_dir: Path | str | None = None,
) -> DataModel:
    """...same docstring, updated Raises..."""
    dir_path = Path(models_dir).resolve() if models_dir else DEFAULT_MODELS_DIR
    cache_key = f"{dir_path}:{model_name}"

    if cache_key in _cache:
        return _cache[cache_key]

    model_path = dir_path / f"{model_name}.yaml"
    if not model_path.exists():
        raise FileNotFoundError(f"Data model '{model_name}' not found at {model_path}")

    try:
        raw = yaml.safe_load(model_path.read_text())
    except yaml.YAMLError as e:
        raise ValueError(f"Invalid model YAML in {model_path}: {e}") from e

    try:
        data_model = DataModel.model_validate(raw)
    except Exception as e:
        raise ValueError(f"Invalid model YAML in {model_path}: {e}") from e

    if data_model.model != model_name:
        raise ValueError(
            f"Model name mismatch: file is '{model_name}.yaml' "
            f"but model field is '{data_model.model}'"
        )

    _cache[cache_key] = data_model
    return data_model
```

**Key:** Failed models are NOT cached — only successfully loaded models enter `_cache`.
